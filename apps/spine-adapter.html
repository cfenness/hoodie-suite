The entire adapter is the snippet below — about ten lines. Each real app drops this in once and inherits shared scope, account, and date basis automatically: 
<script src="../spine/spine.js"></script>
<script>
  const spine = HoodieSpine.connect({
    name: "My App",
    onContext: (ctx) => applyContext(ctx)   // filter your views to ctx.scope / ctx.account / ctx.dateRange
  });
  // to push a change up:   spine.setContext({ dateRange:{ basis:"YTD" } });
  // to deep-link an app:   spine.navigate("dashboard", ctx.scope);
</script>
